Option Explicit

Function RunShellCommand(command As String) As String
    Dim shell As Object
    Set shell = CreateObject("WScript.Shell")

    Dim output As String
    output = shell.Run("powershell.exe -Command " & command, 0)

    RunShellCommand = output
End Function
